    |  * JSONGetValuesAtPath ( array ; path)

    |  * RETURNS:
    |  *	a list containing only the values at that path
    |  *
    |  * PARAMETERS:

    | 
    |

    | Let ( [
    | 
    | 	notValid = Left ( JSONFormatElements ( array ) ; 1 ) = "?" ;
    | 	c = ValueCount ( JSONListValues ( array ; "" ) ) ;
    | 	
    | 	thisItem = JSONGetElement ( array ;  c - 1  ) ;
