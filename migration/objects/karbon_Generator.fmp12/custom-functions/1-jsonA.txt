    | Let ( [
    | 	arrayValues =
    | 	If (
    | 		Right ( arrayValues ; 1 ) = "," ;
    | 		Left ( arrayValues ; Length ( arrayValues ) - 1 ) ;
    | 		arrayValues
    | 	)
