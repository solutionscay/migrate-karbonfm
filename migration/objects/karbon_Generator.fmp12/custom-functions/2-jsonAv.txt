    |  * jsonAv( value )

    |  * RETURNS:
    |  *	prepares a string for inclusion as an element in a JSON array
    |  *
    |  * PARAMETERS:

    | 	valueIsJson = firstAndLast = "{}" or firstAndLast = "[]" ;

    | 		) ;
