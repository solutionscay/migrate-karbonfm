CustomFunction id=24 functionArity=1 visible=True parameters=pluginName name=design_PluginStatus
  Calculation:
    | Let(
    | [
    | pluginList = Get (InstalledFMPlugins);
    | l = Length(pluginName)+1 ;
    | expression = "Left(item ; " & l & ")="& Quote(pluginName& ";")  ;
    | 
    | pluginData = FilterValuesByExpression ( pluginList ; expression );
    | pluginDataSplit = Substitute(pluginData; ";"; "¶");
    | status = GetValue(pluginDataSplit ; 3 );
    | 
    | status = Case( IsEmpty( status ) ; "Not Installed" ; status )
    | 
    | ];
    | 
    | status
    | 
    | )
